AIM

----- File name: process_sort.c -----
#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <unistd.h>

#define MAX 50

void sort_ascending(int a[], int n) {
    int i, j, temp;
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if (a[j] > a[j + 1]) {
                temp = a[j]; a[j] = a[j + 1]; a[j + 1] = temp;
            }
}

void sort_descending(int a[], int n) {
    int i, j, temp;
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if (a[j] < a[j + 1]) {
                temp = a[j]; a[j] = a[j + 1]; a[j + 1] = temp;
            }
}

void print_array(const char *label, int a[], int n) {
    int i;
    printf("%s", label);
    for (i = 0; i < n; i++) printf("%s%d", i ? " " : "", a[i]);
    printf("\n");
}

int main(void) {
    int values[MAX], n, i;
    pid_t pid;

    printf("Enter number of integers (1-%d): ", MAX);
    if (scanf("%d", &n) != 1 || n < 1 || n > MAX) {
        fprintf(stderr, "Invalid number of integers.\n");
        return 1;
    }
    printf("Enter %d integers: ", n);
    for (i = 0; i < n; i++) {
        if (scanf("%d", &values[i]) != 1) {
            fprintf(stderr, "Invalid integer input.\n");
            return 1;
        }
    }

    pid = fork();
    if (pid < 0) {
        perror("fork");
        return 1;
    }

    if (pid == 0) {
        sort_descending(values, n);
        print_array("Child sorted descending: ", values, n);
        fflush(stdout);
        _exit(0);
    } else {
        int status;
        sort_ascending(values, n);
        print_array("Parent sorted ascending: ", values, n);
        waitpid(pid, &status, 0);
        printf("Parent: child process has finished.\n");
    }
    return 0;
}

----- File name: zombie_demo.c -----
#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <unistd.h>

int main(void) {
    pid_t pid = fork();
    if (pid < 0) { perror("fork"); return 1; }

    if (pid == 0) {
        printf("Child (PID %d) exits now.\n", getpid());
        fflush(stdout);
        _exit(0);
    }

    printf("Parent (PID %d) sleeping for 15 seconds before wait().\n", getpid());
    printf("During this time, inspect the child using: ps -o pid,ppid,stat,cmd -p %d\n", pid);
    fflush(stdout);
    sleep(15);
    waitpid(pid, NULL, 0);
    printf("Parent called wait(); zombie entry is now collected.\n");
    return 0;
}

----- File name: orphan_demo.c -----
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int main(void) {
    pid_t pid = fork();
    if (pid < 0) { perror("fork"); return 1; }

    if (pid > 0) {
        printf("Parent (PID %d) exits; child PID is %d.\n", getpid(), pid);
        fflush(stdout);
        return 0;
    }

    printf("Child started: PID=%d, PPID=%d\n", getpid(), getppid());
    fflush(stdout);
    sleep(3);
    printf("Child after parent exit: PID=%d, new PPID=%d\n", getpid(), getppid());
    return 0;
}

======================================================================
NOTES
======================================================================
Zombie  = child finished, but parent has not called wait() yet (STAT shows Z / <defunct>).
Orphan  = parent finished before child; child is adopted by init/systemd (PPID = 1; on some Ubuntu desktops it may be another PID like your user systemd).
PIDs in your output will be different - that is normal.
