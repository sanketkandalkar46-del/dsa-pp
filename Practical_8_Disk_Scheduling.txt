----- File name: disk_scheduling.c -----
#include <stdio.h>
#include <stdlib.h>

#define MAX 50

int n, head, size;
int req[MAX];

void sort_requests(int s[]) {
    for (int i = 0; i < n; i++) s[i] = req[i];
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - i - 1; j++)
            if (s[j] > s[j + 1]) {
                int t = s[j]; s[j] = s[j + 1]; s[j + 1] = t;
            }
}

void sstf(void) {
    int done[MAX] = {0};
    int cur = head, total = 0;
    printf("\nSSTF\nOrder: %d", head);
    for (int k = 0; k < n; k++) {
        int min = 1000000, idx = -1;
        for (int i = 0; i < n; i++) {
            int distance = abs(req[i] - cur);
            if (!done[i] && distance < min) {
                min = distance;
                idx = i;
            }
        }
        done[idx] = 1;
        total += min;
        cur = req[idx];
        printf(" -> %d", cur);
    }
    printf("\nTotal head movement = %d\n", total);
}

/* Moving away from spindle means toward larger cylinder numbers. */
void scan(void) {
    int s[MAX], cur = head, total = 0;
    sort_requests(s);
    printf("\nSCAN\nOrder: %d", head);

    for (int i = 0; i < n; i++) {
        if (s[i] >= head) {
            total += s[i] - cur;
            cur = s[i];
            printf(" -> %d", cur);
        }
    }
    if (cur != size - 1) {
        total += (size - 1) - cur;
        cur = size - 1;
        printf(" -> %d", cur);
    }
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] < head) {
            total += cur - s[i];
            cur = s[i];
            printf(" -> %d", cur);
        }
    }
    printf("\nTotal head movement = %d\n", total);
}

void clook(void) {
    int s[MAX], cur = head, total = 0, first_lower = -1;
    sort_requests(s);
    printf("\nC-LOOK\nOrder: %d", head);

    for (int i = 0; i < n; i++) {
        if (s[i] >= head) {
            total += s[i] - cur;
            cur = s[i];
            printf(" -> %d", cur);
        }
    }

    for (int i = 0; i < n; i++)
        if (s[i] < head) { first_lower = i; break; }

    if (first_lower != -1) {
        int highest_lower = -1;
        for (int i = 0; i < n; i++)
            if (s[i] < head) highest_lower = i;

        /* C-LOOK jump from highest request to lowest request. */
        total += abs(cur - s[first_lower]);
        cur = s[first_lower];
        printf(" -> %d", cur);

        for (int i = first_lower + 1; i <= highest_lower; i++) {
            total += s[i] - cur;
            cur = s[i];
            printf(" -> %d", cur);
        }
    }
    printf("\nTotal head movement = %d\n", total);
}

int main(void) {
    printf("Enter total number of cylinders: ");
    if (scanf("%d", &size) != 1 || size < 1) {
        printf("Invalid cylinder count.\n");
        return 1;
    }
    printf("Enter number of requests: ");
    if (scanf("%d", &n) != 1 || n < 1 || n > MAX) {
        printf("Invalid number of requests (1 to %d).\n", MAX);
        return 1;
    }
    printf("Enter the request queue: ");
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &req[i]) != 1 || req[i] < 0 || req[i] >= size) {
            printf("Each request must be between 0 and %d.\n", size - 1);
            return 1;
        }
    }
    printf("Enter initial head position: ");
    if (scanf("%d", &head) != 1 || head < 0 || head >= size) {
        printf("Head must be between 0 and %d.\n", size - 1);
        return 1;
    }
    sstf();
    scan();
    clook();
    return 0;
}
