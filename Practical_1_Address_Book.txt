----- File name: addressbook.sh -----
#!/usr/bin/env bash

DATA_FILE="addressbook.txt"

create_book() {
    if [[ -e "$DATA_FILE" ]]; then
        read -r -p "File exists. Replace it? (y/n): " answer
        [[ "$answer" == "y" || "$answer" == "Y" ]] || return
    fi
    printf "NAME | PHONE | EMAIL\n" > "$DATA_FILE"
    echo "Address book created: $DATA_FILE"
}

view_book() {
    if [[ ! -f "$DATA_FILE" ]]; then
        echo "Address book does not exist. Choose option 1 first."
        return
    fi
    cat "$DATA_FILE"
}

insert_record() {
    if [[ ! -f "$DATA_FILE" ]]; then
        echo "Address book does not exist. Choose option 1 first."
        return
    fi
    read -r -p "Enter name: " name
    read -r -p "Enter phone: " phone
    read -r -p "Enter email: " email
    printf "%s | %s | %s\n" "$name" "$phone" "$email" >> "$DATA_FILE"
    echo "Record inserted."
}

delete_record() {
    if [[ ! -f "$DATA_FILE" ]]; then
        echo "Address book does not exist. Choose option 1 first."
        return
    fi
    read -r -p "Enter the exact name to delete: " name
    tmp_file=$(mktemp)
    awk -F' \\| ' -v target="$name" 'NR == 1 || $1 != target' "$DATA_FILE" > "$tmp_file"
    if cmp -s "$DATA_FILE" "$tmp_file"; then
        echo "No matching record found."
        rm -f "$tmp_file"
    else
        mv "$tmp_file" "$DATA_FILE"
        echo "Record deleted."
    fi
}

modify_record() {
    if [[ ! -f "$DATA_FILE" ]]; then
        echo "Address book does not exist. Choose option 1 first."
        return
    fi
    read -r -p "Enter the exact name to modify: " old_name
    if ! awk -F' \\| ' -v target="$old_name" 'NR > 1 && $1 == target {found=1} END {exit !found}' "$DATA_FILE"; then
        echo "No matching record found."
        return
    fi
    read -r -p "Enter new name: " new_name
    read -r -p "Enter new phone: " new_phone
    read -r -p "Enter new email: " new_email
    tmp_file=$(mktemp)
    awk -F' \\| ' -v old="$old_name" -v new="$new_name" -v phone="$new_phone" -v email="$new_email" '
        NR == 1 {print; next}
        $1 == old {printf "%s | %s | %s\n", new, phone, email; next}
        {print}
    ' "$DATA_FILE" > "$tmp_file"
    mv "$tmp_file" "$DATA_FILE"
    echo "Record modified."
}

while true; do
    echo
    echo "===== ADDRESS BOOK ====="
    echo "1. Create address book"
    echo "2. View address book"
    echo "3. Insert a record"
    echo "4. Delete a record"
    echo "5. Modify a record"
    echo "6. Exit"
    read -r -p "Enter choice: " choice
    case "$choice" in
        1) create_book ;;
        2) view_book ;;
        3) insert_record ;;
        4) delete_record ;;
        5) modify_record ;;
        6) echo "Exiting address book."; break ;;
        *) echo "Invalid choice. Enter a number from 1 to 6." ;;
    esac
done
