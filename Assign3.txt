# Function for Selection Sort
def selection_sort(a):
    n = len(a)
    for i in range(n):
        min_index = i
        for j in range(i + 1, n):
            if a[j] < a[min_index]:
                min_index = j

        # Swap
        a[i], a[min_index] = a[min_index], a[i]

# Function for Quick Sort
def quick_sort(a):
    if len(a) <= 1:
        return a

    pivot = a[0]
    left = []
    right = []

    for i in range(1, len(a)):
        if a[i] <= pivot:
            left.append(a[i])
        else:
            right.append(a[i])

    return quick_sort(left) + [pivot] + quick_sort(right)

# Main Program
n = int(input("Enter number of students: "))
marks = []

print("Enter percentage of students:")
for i in range(n):
    percentage = float(input())
    marks.append(percentage)

print("\nOriginal Array:")
print(marks)

print("\n--- SELECTION SORT ---")
selection_array = marks.copy()
selection_sort(selection_array)

print("Sorted Array:")
print(selection_array)

print("\nTop Five Scores:")
top_five = selection_array[-5:]
for score in reversed(top_five):
    print(score)

print("\n--- QUICK SORT ---")
quick_array = quick_sort(marks)

print("Sorted Array:")
print(quick_array)

print("\nTop Five Scores:")
top_five = quick_array[-5:]
for score in reversed(top_five):
    print(score)
