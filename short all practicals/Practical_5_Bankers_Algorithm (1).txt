----- File name: bankers.c -----
#include <stdio.h>
int main() {
    int n, m, i, j, k, c = 0, al[10][10], mx[10][10], av[10], f[10] = {0}, s[10];
    printf("Processes and Resources: "); scanf("%d %d", &n, &m);
    printf("Allocation:\n"); for (i = 0; i < n; i++) for (j = 0; j < m; j++) scanf("%d", &al[i][j]);
    printf("Max:\n");        for (i = 0; i < n; i++) for (j = 0; j < m; j++) scanf("%d", &mx[i][j]);
    printf("Available:\n");  for (j = 0; j < m; j++) scanf("%d", &av[j]);

    for (k = 0; k < n; k++)
        for (i = 0; i < n; i++) {
            if (f[i]) continue;
            for (j = 0; j < m && mx[i][j] - al[i][j] <= av[j]; j++);   /* need <= available? */
            if (j == m) {
                for (j = 0; j < m; j++) av[j] += al[i][j];
                f[i] = 1; s[c++] = i;
            }
        }
    if (c < n) { printf("UNSAFE state\n"); return 0; }
    printf("SAFE state. Sequence: ");
    for (i = 0; i < n; i++) printf("P%d ", s[i]);
    printf("\n");
}
