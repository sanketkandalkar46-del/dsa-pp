======================================================================
STEP 2: CODE
======================================================================

----- File name: addressbook.sh -----
#!/bin/bash
F=addressbook.txt
while true; do
  echo -e "\n1.Create 2.View 3.Insert 4.Delete 5.Modify 6.Exit"
  read -p "Choice: " c
  case $c in
    1) echo "NAME | PHONE | EMAIL" > $F; echo "Created" ;;
    2) cat $F ;;
    3) read -p "Name: " n; read -p "Phone: " p; read -p "Email: " e
       echo "$n | $p | $e" >> $F; echo "Inserted" ;;
    4) read -p "Name to delete: " n
       grep -v "^$n |" $F > t.txt; mv t.txt $F; echo "Deleted" ;;
    5) read -p "Name to modify: " n; read -p "New name: " a
       read -p "New phone: " p; read -p "New email: " e
       sed -i "s/^$n |.*/$a | $p | $e/" $F; echo "Modified" ;;
    6) break ;;
    *) echo "Invalid choice" ;;
  esac
done
