PRACTICAL 4: Deadlock Avoidance - Dining Philosophers (Semaphores)

======================================================================
AIM
======================================================================
Deadlock-free Dining Philosophers: allow at most N-1 philosophers to try to eat at the same time (room semaphore).

======================================================================

======================================================================
STEP 2: CODE
======================================================================

----- File name: dining_philosophers.c -----
#include <pthread.h>
#include <semaphore.h>
#include <stdio.h>
#include <unistd.h>
#define N 5
sem_t f[N], room;       /* f = forks, room allows only N-1 philosophers */

void *phil(void *a) {
    int i = *(int *)a;
    for (int r = 0; r < 2; r++) {
        printf("Philosopher %d thinking\n", i + 1); usleep(100000);
        sem_wait(&room); sem_wait(&f[i]); sem_wait(&f[(i + 1) % N]);
        printf("Philosopher %d eating\n", i + 1); usleep(100000);
        sem_post(&f[(i + 1) % N]); sem_post(&f[i]); sem_post(&room);
    }
    return 0;
}
int main() {
    pthread_t t[N]; int id[N];
    sem_init(&room, 0, N - 1);
    for (int i = 0; i < N; i++) sem_init(&f[i], 0, 1);
    for (int i = 0; i < N; i++) { id[i] = i; pthread_create(&t[i], 0, phil, &id[i]); }
    for (int i = 0; i < N; i++) pthread_join(t[i], 0);
    printf("All done\n");
}
