#include <iostream>
#include <stack>
#include <string>
using namespace std;

void reverseString(string str) {
    stack<char> s;

    for (char ch : str)
        s.push(ch);

    cout << "Original String: " << str << endl;

    cout << "Reversed String: ";
    while (!s.empty()) {
        cout << s.top();
        s.pop();
    }

    cout << endl;
}

void checkPalindrome(string str) {
    stack<char> s;

    for (char ch : str)
        s.push(ch);

    string rev = "";

    while (!s.empty()) {
        rev += s.top();
        s.pop();
    }

    if (str == rev)
        cout << "The string is a Palindrome." << endl;
    else
        cout << "The string is not a Palindrome." << endl;
}

int main() {
    string str;

    cout << "Enter a string: ";
    cin >> str;

    cout << "\n--- String Reversal ---" << endl;
    reverseString(str);

    cout << "\n--- Palindrome Check ---" << endl;
    checkPalindrome(str);

    return 0;
}
